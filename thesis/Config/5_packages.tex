% ===== Espaçamento 1,15 sem depender do setspace =====
\makeatletter
\AtBeginDocument{%
  \linespread{1.15}\selectfont % aplica 1,15 de forma estável
  % memoir defaults to \flushbottom (twoside), which stretches paragraph and
  % float glue on every page to reach the bottom margin -- the source of the
  % large, uneven gaps between paragraphs, figures and tables.
  \raggedbottom
}

% share the page where the next section starts, so a tall float does not leave
% the rest of the page empty (e.g. the chapter-opening page of ch. 4).

% ===== Folha de rosto dos anexos =====
% \ntprintannexes só acrescenta "Anexos" ao sumário; não há página que marque o
% início do pós-texto. A divisória carrega a entrada do sumário (com número de
% página e link, que o \addappheadtotoc da classe deixa vazios).
\newcommand*{\ntannexdivider}{%
  \clearforchapter
  \thispagestyle{empty}%
  \phantomsection\addcontentsline{toc}{chapter}{\appendixtocname}%
  \vspace*{\fill}%
  \begin{center}
    {\Huge\bfseries\appendixtocname\par}%
  \end{center}
  \vspace*{\fill}%
  \clearforchapter
}
\patchcmd{\ntprintannexes}{\annex\addappheadtotoc}{\annex\ntannexdivider}{}%
  {\PackageWarningNoLine{5_packages}{could not add the annex divider page}}
